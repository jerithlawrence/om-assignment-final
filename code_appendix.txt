===== scripts/exact_method.py =====
"""Held–Karp dynamic programming for small symmetric TSP instances."""

import numpy as np


def held_karp(distance):
    """Return a globally optimal permutation and its closed-tour cost.

    State (mask, j) is a shortest path from node 0 through exactly mask,
    ending at j + 1. Bits represent nodes 1 through n - 1.
    Time O(n^2 2^n); space O(n 2^n). Limit n to protect memory/runtime.
    """
    n = len(distance)
    if not 3 <= n <= 20:
        raise ValueError("Held–Karp supports 3 to 20 nodes in this project")
    m = n - 1
    states = 1 << m
    infinity = np.iinfo(np.int64).max // 4
    cost = np.full((states, m), infinity, dtype=np.int64)
    parent = np.full((states, m), -1, dtype=np.int8)
    for j in range(m):
        cost[1 << j, j] = distance[0, j + 1]
    for mask in range(1, states):
        remaining = mask
        while remaining:
            bit = remaining & -remaining
            j = bit.bit_length() - 1
            previous = mask ^ bit
            if previous:
                candidates = previous
                best, predecessor = infinity, -1
                while candidates:
                    kbit = candidates & -candidates
                    k = kbit.bit_length() - 1
                    value = cost[previous, k] + distance[k + 1, j + 1]
                    if value < best:
                        best, predecessor = value, k
                    candidates ^= kbit
                cost[mask, j] = best
                parent[mask, j] = predecessor
            remaining ^= bit
    last_costs = cost[-1] + distance[1:, 0]
    j = int(np.argmin(last_costs))
    optimum = int(last_costs[j])
    mask, reverse_path = states - 1, []
    while mask:
        reverse_path.append(j + 1)
        predecessor = int(parent[mask, j])
        mask ^= 1 << j
        j = predecessor
    return np.array([0] + reverse_path[::-1]), optimum

===== scripts/heuristic_method.py =====
"""Random-start nearest neighbour followed by best-improvement 2-opt."""

import numpy as np


def tour_length(route, distance):
    """Include the last-to-first edge; route itself contains no repeat."""
    route = np.asarray(route, dtype=int)
    return int(distance[route, np.roll(route, -1)].sum())


def local_search(distance, seed):
    """Return a feasible 2-opt local optimum, initial cost and cost history.

    Only the start is random. Nearest-neighbour ties use smallest index;
    equal 2-opt improvements use row-major order. No iteration/time cap.
    """
    n = len(distance)
    rng = np.random.default_rng(seed)
    route = [int(rng.integers(n))]
    unused = set(range(n)) - set(route)
    while unused:
        node = min(unused, key=lambda j: (distance[route[-1], j], j))
        route.append(node)
        unused.remove(node)
    route = np.array(route)
    history = [tour_length(route, distance)]
    # Nonadjacent edge pairs only; first and last edges are adjacent too.
    i, j = np.triu_indices(n, k=2)
    valid = ~((i == 0) & (j == n - 1))
    i, j = i[valid], j[valid]
    while len(i):
        successor = np.roll(route, -1)
        delta = (distance[route[i], route[j]]
                 + distance[successor[i], successor[j]]
                 - distance[route[i], successor[i]]
                 - distance[route[j], successor[j]])
        move = int(np.argmin(delta))
        if delta[move] >= 0:
            break
        left, right = int(i[move]) + 1, int(j[move]) + 1
        route[left:right] = route[left:right][::-1]
        history.append(history[-1] + int(delta[move]))
    assert tour_length(route, distance) == history[-1]
    return route, history[0], history

"""Reproduce the pcb442 experiments and figures from the local dataset."""

import csv
import hashlib

import platform

matplotlib.use("Agg")

ROOT = Path(__file__).resolve().parents[1]
SIZES = (8, 10, 12, 14, 16, 18, 50, 100, 200, 442)
SEEDS = tuple(range(10))
SUBSET_SEED = 2026
EXACT_REPEATS = 3
FULL_OPTIMUM = 50778  # Published by TSPLIB; not computed by our exact solver.


def load_dataset(path):
    """Read and validate the original TSPLIB coordinate file."""
    metadata, rows = {}, []
    coordinates = False
    for line in path.read_text().splitlines():
        line = line.strip()
        if line == "NODE_COORD_SECTION":
            coordinates = True
        elif line == "EOF":
            break
        elif coordinates and line:
            node, x, y = line.split()
            rows.append((int(node), float(x), float(y)))
        elif ":" in line:
            key, value = line.split(":", 1)
            metadata[key.strip()] = value.strip()
    if (metadata.get("NAME") != "pcb442"
            or metadata.get("TYPE") != "TSP"
            or metadata.get("EDGE_WEIGHT_TYPE") != "EUC_2D"
            or int(metadata.get("DIMENSION", 0)) != 442
            or len(rows) != 442):
        raise ValueError("Expected original pcb442 EUC_2D data")
    if [row[0] for row in rows] != list(range(1, 443)):
        raise ValueError("Expected unique consecutive TSPLIB node IDs")
    xy = np.array([row[1:] for row in rows])
    if not np.isfinite(xy).all():
        raise ValueError("Coordinates must be finite")
    return metadata, xy


def distance_matrix(xy):
    """TSPLIB EUC_2D uses nearest integer, with halves rounded upward."""
    difference = xy[:, None, :] - xy[None, :, :]
    return np.floor(np.sqrt((difference ** 2).sum(axis=2)) + 0.5
                    ).astype(np.int64)


def validate_tour(route, distance, expected):
    if sorted(route) != list(range(len(distance))):
        raise ValueError("Tour must visit every node exactly once")
    if tour_length(route, distance) != expected:
        raise ValueError("Closed-tour cost does not match")


def write_csv(path, rows):
    with path.open("w", newline="") as file:
        writer = csv.DictWriter(file, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def summarize(rows):
    summary = []
    for n in SIZES:
        for method in ("held_karp", "two_opt"):
            group = [r for r in rows if r["n"] == n
                     and r["method"] == method]
            if not group:
                continue
            costs = [r["distance"] for r in group]
            times = [r["runtime_seconds"] for r in group]
            reference = group[0]["reference_distance"]
            summary.append(dict(
                n=n, method=method, runs=len(group), best=min(costs),
                mean=float(np.mean(costs)), worst=max(costs),
                sd=float(np.std(costs, ddof=1)),
                mean_seconds=float(np.mean(times)),
                sd_seconds=float(np.std(times, ddof=1)),
                reference_distance=reference,
                mean_gap_percent=(100 * (np.mean(costs) / reference - 1)
                                  if reference else ""),
                feasible_runs=sum(r["feasible"] for r in group)))
    return summary


def figures(xy, rows, summary, best_route, history, output):
    plt.rcParams.update({"font.size": 10, "axes.spines.top": False,

    def save(name):

        save(name)
    fig, axes = plt.subplots(1, 2, figsize=(10, 4))
    for method, label in (("held_karp", "Held–Karp"),

        group = [r for r in summary if r["method"] == method]
        axes[0].plot([r["n"] for r in group],

    axes[0].legend()
    small = [r for r in summary if r["method"] == "two_opt" and r["n"] <= 18]
    axes[1].bar([str(r["n"]) for r in small],
                [r["mean_gap_percent"] for r in small], color="#28638a")

    full = [r for r in rows if r["n"] == 442]

    ax.set(xlabel="Random seed", ylabel="Closed-tour distance", xticks=SEEDS)

    save("full_benchmark_runs.png")
    fig, ax = plt.subplots(figsize=(8, 3.5))
    ax.plot(range(len(history)), history, color="#28638a")

def main():
    output = ROOT / "output"
    output.mkdir(exist_ok=True)
    dataset = ROOT / "dataset" / "pcb442.tsp"
    metadata, xy = load_dataset(dataset)
    full_distance = distance_matrix(xy)
    order = np.random.default_rng(SUBSET_SEED).permutation(len(xy))
    rows, routes, subsets = [], {}, {}
    best_full, best_history, best_cost = None, None, float("inf")
    started = datetime.now(timezone.utc).isoformat()
    for n in SIZES:
        # Nested subsets; full instance retains the original TSPLIB order.
        indices = np.sort(order[:n]) if n < 442 else np.arange(442)
        subsets[str(n)] = (indices + 1).tolist()
        d = full_distance[np.ix_(indices, indices)]
        reference = FULL_OPTIMUM if n == 442 else None
        jobs = ([('held_karp', run) for run in range(EXACT_REPEATS)]
                if n <= 18 else [])
        jobs += [('two_opt', seed) for seed in SEEDS]
        for method, run in jobs:
            start = perf_counter()
            if method == "held_karp":
                route, cost = held_karp(d)
                initial, history = None, []
            else:
                route, initial, history = local_search(d, run)
                cost = history[-1]
            elapsed = perf_counter() - start
            validate_tour(route, d, cost)
            if method == "held_karp":
                reference = cost
            if reference is not None and cost < reference:
                raise ValueError("Cost is below verified reference")
            key = f"{n}_{method}_{run}"
            routes[key] = (indices[route] + 1).tolist()
            rows.append(dict(
                n=n, method=method, run=run,
                seed=run if method == "two_opt" else "",
                distance=cost, initial_distance=initial,
                runtime_seconds=elapsed, feasible=True,
                status="optimal" if method == "held_karp" else "local_optimum",
                reference_distance=reference,
                gap_percent=100 * (cost / reference - 1) if reference else "",
                accepted_moves=len(history) - 1 if history else ""))
            if n == 442 and cost < best_cost:
                best_full, best_history, best_cost = route.copy(), history, cost
        print(f"Completed n={n}", flush=True)
    summary = summarize(rows)
    write_csv(output / "results.csv", rows)
    write_csv(output / "summary.csv", summary)
    write_csv(output / "convergence.csv",
              [dict(move=i, distance=c) for i, c in enumerate(best_history)])
    (output / "routes.json").write_text(json.dumps(routes, indent=2) + "\n")
    details = dict(
        started_utc=started, python=sys.version, platform=platform.platform(),
        machine=platform.machine(), numpy=np.__version__,
        matplotlib=matplotlib.__version__, dataset=metadata,
        dataset_sha256=hashlib.sha256(dataset.read_bytes()).hexdigest(),
        source_sha256={p.name: hashlib.sha256(p.read_bytes()).hexdigest()
                       for p in sorted((ROOT / "scripts").glob("*.py"))},
        subset_seed=SUBSET_SEED, subset_node_ids=subsets, seeds=SEEDS,
        exact_repeats=EXACT_REPEATS, exact_limit=18,
        heuristic="random-start NN, best-improvement 2-opt until no gain",
        tie_breaking="smallest node index / first row-major edge pair",
        timing="perf_counter; solver only, including initialization; serial",
        published_full_optimum=FULL_OPTIMUM,
        reference_url="https://comopt.ifi.uni-heidelberg.de/software/TSPLIB95/"
                      "tsp/TSP-BEST.html")
    (output / "experiment_details.json").write_text(
        json.dumps(details, indent=2) + "\n")
    figures(xy, rows, summary, best_full, best_history, output)
    print(f"Best full tour: {best_cost}; all {len(rows)} tours feasible")


if __name__ == "__main__":
    main()

===== scripts/verify.py =====
"""Independent correctness checks; run before the experiment pipeline."""

from itertools import permutations

import numpy as np

from exact_method import held_karp
from heuristic_method import local_search, tour_length
from main import ROOT, distance_matrix, load_dataset, validate_tour


def main():
    _, xy = load_dataset(ROOT / "dataset" / "pcb442.tsp")
    assert xy.shape == (442, 2)
    distance = distance_matrix(xy)
    assert np.array_equal(distance, distance.T)
    assert np.all(np.diag(distance) == 0)
    # Half-up rounding must differ from Python's banker's rounding at 2.5.
    assert distance_matrix(np.array([[0, 0], [2.5, 0]]))[0, 1] == 3
    rng = np.random.default_rng(123)
    for n in (4, 6, 8):
        d = distance_matrix(rng.uniform(0, 100, (n, 2)))
        brute = min(tour_length((0,) + p, d)
                    for p in permutations(range(1, n)))
        route, optimum = held_karp(d)
        validate_tour(route, d, optimum)
        assert optimum == brute
        for seed in range(5):
            route, initial, history = local_search(d, seed)
            validate_tour(route, d, history[-1])
            assert initial >= history[-1] >= optimum
            assert all(a > b for a, b in zip(history, history[1:]))
            # Exhaustively verify termination with no improving 2-opt move.
            for i in range(n):
                for j in range(i + 2, n):
                    trial = route.copy()
                    trial[i + 1:j + 1] = trial[i + 1:j + 1][::-1]
                    assert tour_length(trial, d) >= history[-1]
    try:
        validate_tour([0, 1, 1, 3], np.zeros((4, 4)), 0)
    except ValueError:
        pass
    else:
        raise AssertionError("Duplicate visit was not rejected")
    print("PASS: rounding, closure, brute-force agreement, feasibility, 2-opt")


if __name__ == "__main__":
    main()

===== scripts/ilp_method.py =====
"""Exact TSP model with SCIP branch-and-bound and subtour constraints.

Connectivity callbacks follow the PySCIPOpt constraint-handler interface:
https://github.com/scipopt/PySCIPOpt/blob/master/tests/test_tsp.py
"""

from time import perf_counter

import numpy as np
from pyscipopt import Conshdlr, Model, SCIP_RESULT, quicksum

from heuristic_method import local_search, tour_length


class Connectivity(Conshdlr):
    """Reject disconnected cycles and add valid subtour inequalities."""

    def __init__(self, n, edges):
        self.n, self.edges, self.cuts = n, edges, 0

    def components(self, solution=None):
        neighbours = [[] for _ in range(self.n)]
        for (a, b), variable in self.edges.items():
            if self.model.getSolVal(solution, variable) > 1e-6:
                neighbours[a].append(b)
                neighbours[b].append(a)
        remaining, groups = set(range(self.n)), []
        while remaining:
            stack = [remaining.pop()]
            group = set(stack)
            while stack:
                for b in neighbours[stack.pop()]:
                    if b in remaining:
                        remaining.remove(b)
                        group.add(b)
                        stack.append(b)
            groups.append(sorted(group))
        return groups

    def conscheck(self, constraints, solution, *args, **kwargs):
        connected = len(self.components(solution)) == 1
        return {"result": (SCIP_RESULT.FEASIBLE if connected
                           else SCIP_RESULT.INFEASIBLE)}

    def consenfolp(self, constraints, *args, **kwargs):
        groups = self.components()
        if len(groups) == 1:
            return {"result": SCIP_RESULT.FEASIBLE}
        for group in groups:
            internal = (self.edges[a, b] for k, a in enumerate(group)
                        for b in group[k + 1:])
            self.model.addCons(quicksum(internal) <= len(group) - 1)
            self.cuts += 1
        return {"result": SCIP_RESULT.CONSADDED}

    def conslock(self, *args):
        # Degree equalities already lock every variable in both directions.
        pass


def solve_ilp(distance, time_limit=300):
    """Return a valid tour and solver evidence; a time limit is not proof.

    All n(n-1)/2 edges are included. Seed-0 NN + 2-opt provides a feasible
    warm start, included in total runtime. No published tour or value is
    supplied to the optimizer. Disconnected cycle covers are never returned.
    """
    start = perf_counter()
    n = len(distance)
    model = Model("pcb_tsp")
    model.hideOutput()
    model.setIntParam("parallel/maxnthreads", 1)
    model.setIntParam("randomization/randomseedshift", 0)
    model.setBoolParam("misc/allowstrongdualreds", False)
    model.setRealParam("limits/time", float(time_limit))
    model.setRealParam("limits/gap", 0.0)
    model.setRealParam("numerics/feastol", 1e-6)
    edges = {(a, b): model.addVar(vtype="B", obj=int(distance[a, b]))
             for a in range(n) for b in range(a + 1, n)}
    incident = [[] for _ in range(n)]
    for (a, b), variable in edges.items():
        incident[a].append(variable)
        incident[b].append(variable)
    for variables in incident:
        model.addCons(quicksum(variables) == 2)
    handler = Connectivity(n, edges)
    model.includeConshdlr(handler, "connectivity", "one complete tour",
                         chckpriority=-10, needscons=False)
    initial, _, _ = local_search(distance, seed=0)
    solution = model.createSol()
    for a, b in zip(initial, np.roll(initial, -1)):
        model.setSolVal(solution, edges[min(a, b), max(a, b)], 1)
    model.addSol(solution)
    model.optimize()
    neighbours = [[] for _ in range(n)]
    for (a, b), variable in edges.items():
        if model.getVal(variable) > 0.5:
            neighbours[a].append(b)
            neighbours[b].append(a)
    if any(len(v) != 2 for v in neighbours):
        raise ValueError("SCIP did not return degree-two edges")
    route, previous, current = [], -1, 0
    while current not in route:
        route.append(current)
        following = next(v for v in neighbours[current] if v != previous)
        previous, current = current, following
    if current != 0 or sorted(route) != list(range(n)):
        raise ValueError("SCIP returned a disconnected or invalid cycle")
    cost = tour_length(route, distance)
    if abs(cost - model.getObjVal()) > 1e-5:
        raise ValueError("SCIP objective differs from integer tour cost")
    evidence = dict(
        distance=cost, initial_distance=tour_length(initial, distance),
        runtime_seconds=perf_counter() - start,
        solver_seconds=model.getSolvingTime(),
        status=str(model.getStatus()), lower_bound=model.getDualbound(),
        solver_gap=model.getGap(), branch_nodes=model.getNNodes(),
        subtour_constraints=handler.cuts, time_limit_seconds=time_limit,
        variables=len(edges), scip_version="{}.{}.{}".format(
            model.getMajorVersion(), model.getMinorVersion(),
            model.getTechVersion()))
    return np.array(route), evidence

def draw_pair(xy, routes, titles, path):
    """Draw every location and edge on identical axes; arrows show direction."""
    fig, axes = plt.subplots(1, 2, figsize=(11, 6.7), sharex=True, sharey=True)
    for ax, route, title in zip(axes, routes, titles):
        route = np.asarray(route)
        route = np.roll(route, -int(np.where(route == 0)[0][0]))
        closed = np.append(route, route[0])
        points = xy[closed]
        ax.plot(points[:, 0], points[:, 1], color="#28638a", lw=0.75)
        ax.scatter(xy[:, 0], xy[:, 1], s=8, color="#172e40", zorder=3)
        ax.scatter(*xy[0], marker="s", s=40, color="#c44b28",
                   label="Common start / return", zorder=4)
        for k in range(0, len(route), max(1, len(route) // 12)):
            a, b = points[k], points[k + 1]
            ax.annotate("", xy=b, xytext=a,
                        arrowprops=dict(arrowstyle="->", color="#c44b28",
                                        lw=0.9, mutation_scale=9))
        ax.set(title=title, xlabel="x (dataset units)", aspect="equal")
        ax.legend(loc="upper left", fontsize=8)
    axes[0].set_ylabel("y (dataset units)")
    fig.tight_layout()
    fig.savefig(path, dpi=220, bbox_inches="tight")
    plt.close(fig)


def edge_set(route):
    return {tuple(sorted((int(a), int(b))))
            for a, b in zip(route, np.roll(route, -1))}

        exact_times = [float(r["runtime_seconds"]) for r in group
                       if r["method"] == "held_karp"]
        b = min(hs, key=lambda r: int(r["distance"]))
        small.append(dict(
            n=n, exact_distance=int(e["distance"]),
            heuristic_best=int(b["distance"]),
            excess=int(b["distance"]) - int(e["distance"]),
            best_gap_percent=100 * (int(b["distance"]) / int(e["distance"]) - 1),
            exact_mean_seconds=float(np.mean(exact_times)),
            heuristic_total_seconds=sum(float(r["runtime_seconds"]) for r in hs)))
        if n == 10:
            ids = subsets["subset_node_ids"]["10"]
            keys = [f"10_{r['method']}_{r['run']}" for r in [e, b]]
            small_routes = [[ids.index(v) for v in saved[k]] for k in keys]
            draw_pair(xy[np.array(ids) - 1], small_routes,
                      [f"Held–Karp optimum: {e['distance']}",
                       f"Best NN + 2-opt: {b['distance']}"],
                      output / "subset_routes_comparison.png")
    write_csv(output / "best_subset_comparison.csv", small)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--plots-only", action="store_true")
    args = parser.parse_args()
    output = ROOT / "output"
    _, xy = load_dataset(ROOT / "dataset" / "pcb442.tsp")
    distance = distance_matrix(xy)
    path = output / "full_comparison.json"
    if args.plots_only:
        record = json.loads(path.read_text())
    else:
        # Cross-check the ILP on independent small cases, including subtours.
        for n in (4, 6, 8, 10):
            sample = np.random.default_rng(n).uniform(0, 100, (n, 2))
            d = distance_matrix(sample)
            route, evidence = solve_ilp(d, time_limit=30)
            _, optimum = held_karp(d)
            validate_tour(route, d, optimum)
            assert evidence["status"] == "optimal"
        d = np.full((6, 6), 10, dtype=int)
        d[:3, :3] = d[3:, 3:] = 1
        np.fill_diagonal(d, 0)
        route, evidence = solve_ilp(d, time_limit=30)
        validate_tour(route, d, 24)
        assert evidence["status"] == "optimal"
        print("ILP checks passed; solving all 442 nodes...", flush=True)
        route, evidence = solve_ilp(distance, time_limit=300)
        runs = [dict(method="ilp", **evidence, node_ids=(route + 1).tolist())]
        print(f"ILP: {evidence['distance']}, {evidence['status']}", flush=True)
        for seed in SEEDS:
            start = perf_counter()
            route, initial, history = local_search(distance, seed)
            elapsed = perf_counter() - start
            validate_tour(route, distance, history[-1])
            runs.append(dict(method="two_opt", seed=seed,
                             distance=history[-1], initial_distance=initial,
                             runtime_seconds=elapsed, status="local_optimum",
                             node_ids=(route + 1).tolist()))
        record = dict(
            runs=runs, platform=platform.platform(), python=platform.python_version(),
            numpy=np.__version__, pyscipopt=pyscipopt.__version__,
            source_sha256={p.name: hashlib.sha256(p.read_bytes()).hexdigest()
                           for p in sorted((ROOT / "scripts").glob("*.py"))},
            dataset_sha256=hashlib.sha256(
                (ROOT / "dataset/pcb442.tsp").read_bytes()).hexdigest(),
            settings=dict(threads=1, randomseedshift=0, relative_gap=0,
                          feasibility_tolerance=1e-6, strong_dual_reductions=False,
                          time_limit_seconds=300, warm_start="seed-0 NN + 2-opt"),
            timing="Serial; includes model, warm start, solve and extraction",
            verification="ILP agrees with Held–Karp on four independent cases; "
                         "two triangles test passed")
        path.write_text(json.dumps(record, indent=2) + "\n")
    export_comparison(record, xy, distance, output)
    print("Saved full-board and subset comparisons", flush=True)
